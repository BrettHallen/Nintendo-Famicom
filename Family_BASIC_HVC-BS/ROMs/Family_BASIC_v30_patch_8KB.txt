// Constants used in the disassembly:
//   bgGetRam        .eq     $6c00  {addr/1024}      ; region to save bg data with BGGET
//   memoryTop       .eq     $6fff                   ; What FRETOP is set to by default

////////////////////////////////////////////
// Increase RAM limit from $6FFF to $7FFF //
////////////////////////////////////////////
// PATCH0
// lda     #>memoryTop
06A4: 6F -> 7F

// PATCH1
// cmp     #$70                    ;arg >= #$7000?
17D8: 70 -> 80

// PATCH2
// @noHiDst        lda     #(>memoryTop)+1
2DC9: 70 -> 80

// PATCH3
// cmp     #>bgGetRam              ;is there room after program for BG data?
31BE: 6C -> 7C     

// PATCH4
// lda     #>bgGetRam
31CB: 6C -> 7C

// PATCH5
// lda     #>bgGetRam
320C: 6C -> 7C

/////////////////
// RESET PATCH //
/////////////////
// jsr     InitPpuApu
00D4: 

///////////////////////////////////////////
// Token Patch                           //
///////////////////////////////////////////
// TokenizeKeyword lda     #<tbl_KeywordTokens ($CCAB in memory, $4CAB in ROM)
13E7: AB -> 

// lda     #>tbl_KeywordTokens
13EB: CC ->

// lda     #<tbl_KeywordTokens     ;set zpTokTblScan to the start of the tokens table
14E1: AB -> 

// lda     #>tbl_KeywordTokens
14E5: CC ->


	TOKEN_PATCH0:	load = SIDE1A, start = $219e+$13e7, type = overwrite;
	TOKEN_PATCH1:	load = SIDE1A, start = $219e+$13eb, type = overwrite;
	TOKEN_PATCH2:	load = SIDE1A, start = $219e+$14e1, type = overwrite;
	TOKEN_PATCH3:	load = SIDE1A, start = $219e+$14e5, type = overwrite;

	FUNCADDR_PATCH4:	load = SIDE1A, start = $219e+$0433, type = overwrite;
	FUNCADDR_PATCH5:	load = SIDE1A, start = $219e+$0438, type = overwrite;
	FUNCADDR_PATCH6:	load = SIDE1A, start = $219e+$1b39, type = overwrite;
	FUNCADDR_PATCH7:	load = SIDE1A, start = $219e+$1b3e, type = overwrite;

	RESET_PATCH:	load = SIDE1A, start = $219e+$00d4, type = overwrite;
	#GAME_PATCH:		load = SIDE1A, start = $219e+$2d58, type = overwrite;
	FDS_PATCH:		load = SIDE1A, start = $219e+$5000, type = overwrite;
