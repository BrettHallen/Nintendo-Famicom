// Instructions for patching Family BASIC V2.1A to 8KB
// This is intended to convert a cartridge, not create Disk BASIC
// (this is already well documented)
// Originally from (Japanese): 
// http://cmpslv3.stars.ne.jp/Konjo/027/027.htm
// Lots of help from the disassembly:
// https://github.com/micahcowan/fbdasm
// Modified by Brett Hallen, Jan 2026 (corrections, comments, offsets)

// Change JSR $8131 to JMP $80F0
　　00BC-00BE: 20 31 81 -> 4C F0 80

// Change JMP $80AD to JMP $B5AF
　　00EE-00EF: AD 80    -> AF B5

// Increase RAM limit from $6FFF to $7FFF
　　00DB     : 70       -> 60
　　00E0     : 70       -> 60
　　00E7     : 70       -> 60
　　00EC     : 70       -> 60
　　00FD     : 70       -> 60
　　0104     : 70       -> 60
　　010B     : 70       -> 60
　　0110     : 70       -> 60
　　0118     : 70       -> 60
　　0210     : 70       -> 60
　　0213     : 70       -> 60
　　038D     : 70       -> 60
　　0390     : 70       -> 60
　　0393     : 70       -> 60
　　03A5     : 70       -> 60

　　0570     : 77       -> 7F
　　025A     : 78       -> 80

// Change CLI to NOP
　　0220     : 58       -> EA

// Change command from "SYSTEM" to "BGTOOL"
　　4151-4156: 53 59 53 54 45 4D ->  42 47 54 4F 4F 4C

// Version change from "A" to "D"
　　4367     : 41       ->  44

	PATCH_0: load = SIDE1A, start = $219e+$00bc, type = overwrite;
		jsr $9228 ; extra patches
		jsr $814b ; for disk saving
		jmp $80f0 ; replaces jsr $8131

// Reset Patch 0   4427
// JSR $CD94 removed with NOPs
　　0427-0429: 20 94 CD -> EA EA EA
